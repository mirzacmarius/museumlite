test("audit: import jobs reveal another tenant's diagnostics", async () => {
 const t=createConvexTest();
 const a=await createOrgUserWithPermissions(t,["collections.create"]);
 const b=await createOrgUserWithPermissions(t,["collections.create"]);
 const jobId=await t.run(ctx=>ctx.db.insert("importJobs",{organizationId:b.organizationId,type:"csv",status:"failed",progress:0,totalItems:1,processedItems:0,errors:["Private donor Alice: accession import failed"],createdBy:b.userId,createdAt:Date.now()}));

 const direct=await stranger.query(api.imports.getImportJob,{jobId});
 const list=await stranger.query(api.imports.listImportJobs,{organizationId:b.organizationId});
 expect(direct?.errors).toContain("Private donor Alice: accession import failed");
 expect(list[0]?.errors).toContain("Private donor Alice: accession import failed");
});
